#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <errno.h>
#include <fcntl.h>
#include <unistd.h>
#include <sys/stat.h>
#include <sys/wait.h>

#define FILE_NAME "file.txt"
#define MSG       "Hello Linux\n"

#define ERR_EXIT(msg)          \
    do {                       \
        perror(msg);           \
        exit(EXIT_FAILURE);    \
    } while (0)

static void demo_file_io(void)
{
    int fd;
    ssize_t nwritten, nread;
    char buf[128];
    struct stat st;

    fd = open(FILE_NAME, O_CREAT | O_RDWR | O_TRUNC, 0644);
    if (fd < 0)
        ERR_EXIT("open");

    nwritten = write(fd, MSG, strlen(MSG));
    if (nwritten < 0)
        ERR_EXIT("write");
    if ((size_t)nwritten != strlen(MSG))
        fprintf(stderr, "warning: short write (%zd of %zu bytes)\n",
                nwritten, strlen(MSG));

    if (lseek(fd, 0, SEEK_SET) < 0)
        ERR_EXIT("lseek");

    nread = read(fd, buf, sizeof(buf) - 1);
    if (nread < 0)
        ERR_EXIT("read");
    buf[nread] = '\0';

    printf("Read back from file: %s", buf);

    if (close(fd) < 0)
        ERR_EXIT("close");

    if (stat(FILE_NAME, &st) < 0)
        ERR_EXIT("stat");

    printf("File Size: %lld bytes\n", (long long)st.st_size);
    printf("Permissions: %o\n", st.st_mode & 0777);
}

static void demo_fork(void)
{
    pid_t pid = fork();

    if (pid < 0)
        ERR_EXIT("fork");

    if (pid == 0) {
        printf("Child Process (pid=%d, parent pid=%d)\n",
               getpid(), getppid());
        exit(EXIT_SUCCESS);
    }

    int status;
    if (waitpid(pid, &status, 0) < 0)
        ERR_EXIT("waitpid");

    if (WIFEXITED(status))
        printf("Parent Process: child (pid=%d) exited with status %d\n",
               pid, WEXITSTATUS(status));
    else
        printf("Parent Process: child (pid=%d) terminated abnormally\n", pid);
}

int main(void)
{
    demo_file_io();
    demo_fork();
    return EXIT_SUCCESS;
}
